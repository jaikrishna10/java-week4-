#include <stdio.h>
#include <stdlib.h>

void compareTriplets(int a_count, int* a, int b_count, int* b, int* result) {
    int alice_score = 0;
    int bob_score = 0;

    for (int i = 0; i < 3; i++) {
        if (a[i] > b[i]) {
            alice_score++;
        } else if (a[i] < b[i]) {
            bob_score++;
        }
    }

    result[0] = alice_score;
    result[1] = bob_score;
}

int main() {
    int a[3];
    int b[3];
    int result[2];

    // Read Alice's scores
    for (int i = 0; i < 3; i++) {
        scanf("%d", &a[i]);
    }

    // Read Bob's scores
    for (int i = 0; i < 3; i++) {
        scanf("%d", &b[i]);
    }

    // Compare ratings
    compareTriplets(3, a, 3, b, result);

    // Output scores separated by a space
    printf("%d %d\n", result[0], result[1]);

    return 0;
}
